cmake_minimum_required(VERSION 3.28)
project(vivo_miuix_shader_fix LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 23)
set(LSPLANT_STANDALONE OFF CACHE BOOL "" FORCE)
set(LSPLANT_BUILD_SHARED OFF CACHE BOOL "" FORCE)
add_subdirectory(vendor/LSPlant/lsplant/src/main/jni lsplant)
add_library(dobby STATIC IMPORTED)
set_target_properties(dobby PROPERTIES IMPORTED_LOCATION "${CMAKE_CURRENT_SOURCE_DIR}/vendor/dobby/prefab/modules/dobby/libs/android.${ANDROID_ABI}/libdobby.a" INTERFACE_INCLUDE_DIRECTORIES "${CMAKE_CURRENT_SOURCE_DIR}/vendor/dobby/prefab/modules/dobby/include")
add_library(vivo_miuix_shader_fix SHARED src/module.cpp)
target_include_directories(vivo_miuix_shader_fix PRIVATE vendor "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_compile_options(vivo_miuix_shader_fix PRIVATE -fvisibility=hidden -Wall -Wextra)
target_link_options(vivo_miuix_shader_fix PRIVATE -Wl,-z,max-page-size=16384 -Wl,--exclude-libs,ALL)
target_link_libraries(vivo_miuix_shader_fix PRIVATE lsplant_static dobby log dl)
